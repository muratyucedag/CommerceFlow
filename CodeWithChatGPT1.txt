namespace CommerceFlow.ConsoleUI.Examples
{
    public class BankAccountManagement
    {
        private decimal _balance;
        private bool _isActive;

        public string AccountOwner { get; private set; }

        public decimal Balance
        {
            get { return _balance; }
        }

        public bool IsActive
        {
            get { return _isActive; }
        }

        public BankAccountManagement(string accountOwner, decimal balance)
        {
            if (string.IsNullOrWhiteSpace(accountOwner))
                throw new ArgumentException("Hesap sahibi boþ olamaz.");

            if (balance < 0)
                throw new ArgumentException("Baþlangýç bakiyesi negatif olamaz.");

            AccountOwner = accountOwner;
            _balance = balance;
            _isActive = true;
        }

        public void Deposit(decimal amount)
        {
            if (!_isActive)
            {
                Console.WriteLine("Hesap kapalý. Ýþlem yapýlamaz.");
                return;
            }

            if (amount <= 0)
            {
                Console.WriteLine("Yatýrýlacak tutar sýfýrdan büyük olmalýdýr.");
                return;
            }

            _balance += amount;

            Console.WriteLine($"Para yatýrýldý. Güncel bakiye: {_balance} TL");
        }

        public void Withdraw(decimal amount)
        {
            if (!_isActive)
            {
                Console.WriteLine("Hesap kapalý. Ýþlem yapýlamaz.");
                return;
            }

            if (amount <= 0)
            {
                Console.WriteLine("Çekilecek tutar sýfýrdan büyük olmalýdýr.");
                return;
            }

            if (amount > _balance)
            {
                Console.WriteLine("Yetersiz bakiye.");
                return;
            }

            _balance -= amount;

            Console.WriteLine($"Para çekildi. Güncel bakiye: {_balance} TL");
        }

        public void CloseAccount()
        {
            _isActive = false;
            Console.WriteLine("Banka hesabý kapatýldý.");
        }

        public void ShowAccountInformation()
        {
            Console.WriteLine($"Hesap Sahibi: {AccountOwner}");
            Console.WriteLine($"Bakiye: {_balance} TL");
            Console.WriteLine($"Hesap Durumu: {(_isActive ? "Aktif" : "Pasif")}");
        }
    }
}

namespace CommerceFlow.ConsoleUI.InheritanceExamples
{
    public class Vehicle
    {
        public string Brand { get; set; } = "";
        public string Model { get; set; } = "";
        public int ProductionYear { get; set; }
        public string PlateNumber { get; set; } = "";
        public int CurrentSpeed { get; private set; }

        public void Start()
        {
            Console.WriteLine($"{Brand} {Model} çalýþtýrýldý.");
        }

        public void Accelerate(int amount)
        {
            if (amount <= 0)
            {
                Console.WriteLine("Hýz artýþý pozitif olmalýdýr.");
                return;
            }

            CurrentSpeed += amount;

            Console.WriteLine($"Güncel hýz: {CurrentSpeed} km/sa");
        }

        public void Brake(int amount)
        {
            if (amount <= 0)
            {
                Console.WriteLine("Fren miktarý pozitif olmalýdýr.");
                return;
            }

            CurrentSpeed = Math.Max(0, CurrentSpeed - amount);

            Console.WriteLine($"Fren yapýldý. Güncel hýz: {CurrentSpeed} km/sa");
        }

        public void Stop()
        {
            CurrentSpeed = 0;
            Console.WriteLine("Araç durduruldu.");
        }

        public void ShowVehicleInformation()
        {
            Console.WriteLine($"Marka: {Brand}");
            Console.WriteLine($"Model: {Model}");
            Console.WriteLine($"Üretim Yýlý: {ProductionYear}");
            Console.WriteLine($"Plaka: {PlateNumber}");
        }
    }
}

namespace CommerceFlow.ConsoleUI.InheritanceExamples
{
    public class Truck : Vehicle
    {
        public decimal LoadCapacity { get; set; }
        public decimal CurrentLoad { get; private set; }

        public void LoadCargo(decimal weight)
        {
            if (weight <= 0)
            {
                Console.WriteLine("Yük aðýrlýðý pozitif olmalýdýr.");
                return;
            }

            if (CurrentLoad + weight > LoadCapacity)
            {
                Console.WriteLine("Yük kapasitesi aþýlýyor!");
                return;
            }

            CurrentLoad += weight;

            Console.WriteLine($"Yükleme tamamlandý. Mevcut yük: {CurrentLoad} ton");
        }

        public void ShowTruckInformation()
        {
            ShowVehicleInformation();

            Console.WriteLine($"Yük Kapasitesi: {LoadCapacity} ton");
            Console.WriteLine($"Mevcut Yük: {CurrentLoad} ton");
        }
    }
}
