namespace CommerceFlow.ConsoleUI.HotelExamples
{
    public class HotelEmployee
    {
        // PUBLIC - Genel personel bilgileri
        public int EmployeeId { get; set; }
        public string FullName { get; set; } = "";
        public string Department { get; set; } = "";
        public string Position { get; set; } = "";
        public DateTime HireDate { get; set; }
        public bool IsActive { get; set; } = true;


        // PRIVATE - Maaþ bilgileri
        private decimal Salary = 45000;
        private decimal Bonus = 0;


        // PROTECTED - Personel çalýþma ve konaklama bilgileri
        protected string StaffRoom = "P-205";
        protected string StaffCorridor = "B Blok - Personel Katý";
        protected string Shift = "08:00 - 16:00";
        protected int WeeklyWorkingHours = 45;
        protected int AnnualLeaveDays = 14;


        // PUBLIC METHODS
        public void ShowEmployeeInformation()
        {
            Console.WriteLine("Personel No: " + EmployeeId);
            Console.WriteLine("Ad Soyad: " + FullName);
            Console.WriteLine("Departman: " + Department);
            Console.WriteLine("Pozisyon: " + Position);
            Console.WriteLine("Ýþe Giriþ Tarihi: " + HireDate.ToShortDateString());
            Console.WriteLine("Durum: " + (IsActive ? "Aktif" : "Pasif"));
        }

        public void ShowSalaryInformation()
        {
            Console.WriteLine("Maaþ: " + Salary + " TL");
            Console.WriteLine("Prim: " + Bonus + " TL");
            Console.WriteLine("Toplam: " + (Salary + Bonus) + " TL");
        }

        public void IncreaseSalary(decimal amount)
        {
            if (amount > 0)
            {
                Salary += amount;
                Console.WriteLine("Maaþ artýþý gerçekleþtirildi.");
            }
        }

        public void AddBonus(decimal amount)
        {
            if (amount > 0)
            {
                Bonus += amount;
                Console.WriteLine("Personel primi eklendi.");
            }
        }


        // PROTECTED METHODS
        protected void ShowAccommodationInformation()
        {
            Console.WriteLine("Personel Odasý: " + StaffRoom);
            Console.WriteLine("Personel Koridoru: " + StaffCorridor);
        }

        protected void ShowWorkingInformation()
        {
            Console.WriteLine("Vardiya: " + Shift);
            Console.WriteLine("Haftalýk Çalýþma: " + WeeklyWorkingHours + " saat");
            Console.WriteLine("Yýllýk Ýzin: " + AnnualLeaveDays + " gün");
        }
    }
}

namespace CommerceFlow.ConsoleUI.HotelExamples
{
    public class HouseKeeper : HotelEmployee
    {
        // Kat hizmetlerine özgü özellikler
        public int AssignedFloor { get; set; }
        public int DailyRoomTarget { get; set; }
        public int CleanedRoomCount { get; private set; }

        // Personel odasý tahsisi
        public void AssignStaffRoom(string roomNumber)
        {
            StaffRoom = roomNumber;
            Console.WriteLine("Personel odasý tahsis edildi.");
        }

        // Vardiya belirleme
        public void AssignShift(string newShift)
        {
            Shift = newShift;
            Console.WriteLine("Vardiya güncellendi.");
        }

        // Oda temizleme iþlemi
        public void CleanRoom(int roomNumber)
        {
            if (roomNumber <= 0)
            {
                Console.WriteLine("Geçersiz oda numarasý.");
                return;
            }

            CleanedRoomCount++;

            Console.WriteLine(
                $"{roomNumber} numaralý oda temizlendi.");
        }

        // Günlük temizlik durumunu görüntüleme
        public void ShowCleaningStatus()
        {
            Console.WriteLine($"Günlük Hedef: {DailyRoomTarget}");
            Console.WriteLine($"Temizlenen Oda: {CleanedRoomCount}");
            Console.WriteLine(
                $"Kalan Oda: {Math.Max(0, DailyRoomTarget - CleanedRoomCount)}");
        }

        // Personel bilgilerini görüntüleme
        public void ShowHouseKeeperInformation()
        {
            ShowEmployeeInformation();

            Console.WriteLine($"Sorumlu Kat: {AssignedFloor}");

            ShowAccommodationInformation();
            ShowWorkingInformation();
        }
    }
}

namespace CommerceFlow.ConsoleUI.HotelExamples
{
    public class SecurityGuard : HotelEmployee
    {
        // Güvenlik görevlisine özgü özellikler
        public string SecurityArea { get; set; } = "";
        public bool HasSecurityCertificate { get; set; }
        public int IncidentCount { get; private set; }

        // Görev bölgesi belirleme
        public void AssignSecurityArea(string area)
        {
            SecurityArea = area;
            Console.WriteLine("Güvenlik bölgesi belirlendi.");
        }

        // Olay kaydý oluþturma
        public void ReportIncident(string description)
        {
            IncidentCount++;

            Console.WriteLine("Güvenlik olayý kaydedildi.");
            Console.WriteLine("Olay: " + description);
            Console.WriteLine("Toplam Olay: " + IncidentCount);
        }

        // Protected alanýn kullanýmý
        public void ChangeShift(string newShift)
        {
            Shift = newShift;
            Console.WriteLine("Güvenlik vardiyasý güncellendi.");
        }

        // Virtual metodun yeniden uygulanmasý
        public override void StartShift()
        {
            Console.WriteLine(
                $"{FullName} güvenlik vardiyasýný baþlattý.");

            Console.WriteLine("Güvenlik ekipmanlarý kontrol edildi.");
        }

        // Polymorphism için override
        public override void PerformDuty()
        {
            Console.WriteLine(
                $"{FullName} güvenlik noktasýna geçti.");

            Console.WriteLine(
                $"{SecurityArea} bölgesinde devriye baþladý.");

            Console.WriteLine(
                "Giriþ ve çýkýþ kontrolleri gerçekleþtiriliyor.");
        }

        // Güvenlik personeli bilgileri
        public void ShowSecurityInformation()
        {
            ShowEmployeeInformation();

            Console.WriteLine("Güvenlik Bölgesi: " + SecurityArea);
            Console.WriteLine(
                "Güvenlik Sertifikasý: " + HasSecurityCertificate);

            Console.WriteLine("Olay Sayýsý: " + IncidentCount);

            ShowWorkingInformation();
        }
    }
}
