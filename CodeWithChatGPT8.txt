using RestaurantFlow.ConsoleUI.ISPExamples.Interfaces;

namespace RestaurantFlow.ConsoleUI.ISPExamples.Employees
{
    public class Waiter : IOrderService
    {
        public string FullName { get; set; } = "";
        public int TotalOrders { get; private set; }
        public int LastOrderTable { get; private set; }

        public void TakeOrder(
            int tableNumber,
            string foodName,
            int quantity)
        {
            Console.WriteLine("===== SÝPARÝÞ ALMA =====");

            if (tableNumber <= 0)
            {
                Console.WriteLine("Geçersiz masa numarasý.");
                return;
            }

            if (string.IsNullOrWhiteSpace(foodName))
            {
                Console.WriteLine("Yemek adý boþ olamaz.");
                return;
            }

            if (quantity <= 0)
            {
                Console.WriteLine("Sipariþ miktarý geçersiz.");
                return;
            }

            TotalOrders++;
            LastOrderTable = tableNumber;

            Console.WriteLine("Garson: " + FullName);
            Console.WriteLine("Masa No: " + tableNumber);
            Console.WriteLine("Yemek: " + foodName);
            Console.WriteLine("Adet: " + quantity);
            Console.WriteLine("Sipariþ mutfaða iletildi.");
            Console.WriteLine("Toplam Sipariþ Sayýsý: " + TotalOrders);
        }

        public void ServeFood(int tableNumber)
        {
            Console.WriteLine("===== YEMEK SERVÝSÝ =====");

            if (tableNumber <= 0)
            {
                Console.WriteLine("Geçersiz masa numarasý.");
                return;
            }

            if (tableNumber != LastOrderTable)
            {
                Console.WriteLine(
                    "Bu masa için kayýtlý sipariþ bulunamadý.");
                return;
            }

            Console.WriteLine("Garson: " + FullName);
            Console.WriteLine("Masa No: " + tableNumber);
            Console.WriteLine("Hazýrlanan yemekler kontrol edildi.");
            Console.WriteLine("Yemekler masaya servis edildi.");
        }
    }
}

using RestaurantFlow.ConsoleUI.ISPExamples.Interfaces;

namespace RestaurantFlow.ConsoleUI.ISPExamples.Employees
{
    public class Cashier : IPaymentService
    {
        public string FullName { get; set; } = "";
        public decimal TotalRevenue { get; private set; }
        public int CompletedPaymentCount { get; private set; }

        private bool _hasUnreceiptedPayment;
        private decimal _lastPaidAmount;

        public bool ProcessPayment(
            decimal amount,
            decimal receivedAmount)
        {
            Console.WriteLine("===== ÖDEME ÝÞLEMÝ =====");

            if (amount <= 0)
            {
                Console.WriteLine("Ödeme tutarý geçersiz.");
                return false;
            }

            if (receivedAmount < amount)
            {
                Console.WriteLine("Alýnan tutar yetersiz.");
                Console.WriteLine(
                    "Eksik Tutar: " + (amount - receivedAmount) + " TL");

                return false;
            }

            if (_hasUnreceiptedPayment)
            {
                Console.WriteLine(
                    "Önce önceki ödemenin fiþini oluþturun.");

                return false;
            }

            decimal change = receivedAmount - amount;

            TotalRevenue += amount;
            CompletedPaymentCount++;
            _lastPaidAmount = amount;
            _hasUnreceiptedPayment = true;

            Console.WriteLine("Kasiyer: " + FullName);
            Console.WriteLine("Ödeme Tutarý: " + amount + " TL");
            Console.WriteLine("Alýnan Tutar: " + receivedAmount + " TL");
            Console.WriteLine("Para Üstü: " + change + " TL");
            Console.WriteLine("Ödeme baþarýyla tamamlandý.");

            return true;
        }

        public void PrintReceipt(int orderId, decimal amount)
        {
            Console.WriteLine("===== ÖDEME FÝÞÝ =====");

            if (!_hasUnreceiptedPayment)
            {
                Console.WriteLine("Fiþ oluþturulacak ödeme bulunamadý.");
                return;
            }

            if (orderId <= 0 || amount != _lastPaidAmount)
            {
                Console.WriteLine("Fiþ bilgileri ödeme ile uyuþmuyor.");
                return;
            }

            Console.WriteLine("RestaurantFlow");
            Console.WriteLine("Sipariþ No: " + orderId);
            Console.WriteLine("Kasiyer: " + FullName);
            Console.WriteLine("Ödenen Tutar: " + amount + " TL");
            Console.WriteLine("Ödeme Durumu: Baþarýlý");
            Console.WriteLine("Fiþ oluþturuldu.");

            _hasUnreceiptedPayment = false;
        }

        public void ShowDailyRevenue()
        {
            Console.WriteLine("===== KASA ÖZETÝ =====");

            Console.WriteLine("Kasiyer: " + FullName);

            Console.WriteLine(
                "Tamamlanan Ödeme: " + CompletedPaymentCount);

            Console.WriteLine(
                "Toplam Tahsilat: " + TotalRevenue + " TL");
        }
    }
}

using RestaurantFlow.ConsoleUI.ISPExamples.Employees;

Waiter waiter = new Waiter();
waiter.FullName = "Mehmet Demir";

Chef chef = new Chef();
chef.FullName = "Ahmet Yýlmaz";
chef.Specialty = "Izgara Yemekleri";

Cashier cashier = new Cashier();
cashier.FullName = "Zeynep Kaya";

Console.WriteLine("=== RESTAURANTFLOW ===");
Console.WriteLine("=== GÜNLÜK RESTORAN OPERASYONLARI ===");

Console.WriteLine();

// 1. Garson sipariþ alýyor.
waiter.TakeOrder(12, "Hamburger Menü", 2);

Console.WriteLine();

// 2. Aþçý stok kontrolü yapýyor.
chef.CheckKitchenStock("Hamburger Ekmeði", 15);

Console.WriteLine();

// 3. Aþçý yemekleri hazýrlýyor.
chef.PrepareFood("Hamburger Menü", 2);

Console.WriteLine();

// 4. Garson yemekleri servis ediyor.
waiter.ServeFood(12);

Console.WriteLine();

// 5. Kasiyer ödeme alýyor.
bool paymentResult = cashier.ProcessPayment(700, 1000);

Console.WriteLine();

// 6. Ödeme baþarýlýysa fiþ oluþturuluyor.
if (paymentResult)
{
    cashier.PrintReceipt(1001, 700);
}

Console.WriteLine();

// 7. Günlük kasa özeti görüntüleniyor.
cashier.ShowDailyRevenue();

Console.ReadLine();
