namespace CommerceFlow.NTierConsoleUI.Helpers
{
    public static class ConsoleDesign
    {
        public static void WriteHeader(string title)
        {
            Console.Clear();

            Console.ForegroundColor = ConsoleColor.Cyan;

            Console.WriteLine("==================================================");
            Console.WriteLine("                 COMMERCE FLOW                    ");
            Console.WriteLine("==================================================");

            Console.ResetColor();

            Console.WriteLine();

            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine("  " + title.ToUpper());
            Console.ResetColor();

            Console.WriteLine();
        }

        public static void WriteMenuItem(int number, string title)
        {
            Console.ForegroundColor = ConsoleColor.Cyan;
            Console.Write("  [" + number + "] ");

            Console.ForegroundColor = ConsoleColor.White;
            Console.WriteLine(title);

            Console.ResetColor();
        }

        public static void WriteSuccess(string message)
        {
            Console.ForegroundColor = ConsoleColor.Green;
            Console.WriteLine("  " + message);
            Console.ResetColor();
        }

        public static void WriteError(string message)
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine("  " + message);
            Console.ResetColor();
        }

        public static void WriteInfo(string message)
        {
            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine("  " + message);
            Console.ResetColor();
        }

        public static void WaitForKey()
        {
            Console.WriteLine();

            Console.ForegroundColor = ConsoleColor.DarkGray;
            Console.WriteLine("  Devam etmek için bir tuþa basýnýz...");
            Console.ResetColor();

            Console.ReadKey(true);
        }
    }
}

public void AddCategory()
{
    ConsoleRestaurantDesign.WriteHeader("Yeni Kategori Ekle");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Kategori Adý: ");
    Console.ResetColor();

    string? categoryName = Console.ReadLine();

    if (string.IsNullOrWhiteSpace(categoryName))
    {
        ConsoleRestaurantDesign.WriteError(
            "Kategori adý boþ býrakýlamaz.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Category category = new Category
    {
        CategoryName = categoryName.Trim()

    ConsoleRestaurantDesign.WaitForKey();
}

public void ShowCategoryDetail()
{
    ConsoleRestaurantDesign.WriteHeader("Kategori Detayý");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Kategori ID: ");
    Console.ResetColor();

    string? input = Console.ReadLine();

    if (!int.TryParse(input, out int categoryId) || categoryId <= 0)
    {
        ConsoleRestaurantDesign.WriteError(
            "Geçerli bir kategori ID deðeri giriniz.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Category? category = _categoryService.GetCategoryById(categoryId);

    if (category == null)
    {
        ConsoleRestaurantDesign.WriteError(
            "Belirtilen ID deðerine ait kategori bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  ---------------------------------------------");
    Console.WriteLine("                 KATEGORÝ BÝLGÝLERÝ");
    Console.WriteLine("  ---------------------------------------------");
    Console.ResetColor();

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Kategori ID   : ");
    Console.ForegroundColor = ConsoleColor.White;
    Console.WriteLine(category.CategoryId);

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Kategori Adý  : ");
    Console.ForegroundColor = ConsoleColor.White;
    Console.WriteLine(category.CategoryName);

    Console.ResetColor();

    Console.WriteLine();

    ConsoleRestaurantDesign.WriteSuccess(
        "Kategori bilgileri baþarýyla getirildi.");

    ConsoleRestaurantDesign.WaitForKey();
}

public void DeleteCategory()
{
    ConsoleRestaurantDesign.WriteHeader("Kategori Silme");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Silinecek Kategori ID: ");
    Console.ResetColor();

    string? input = Console.ReadLine();

    if (!int.TryParse(input, out int categoryId) || categoryId <= 0)
    {
        ConsoleRestaurantDesign.WriteError(
            "Geçerli bir kategori ID deðeri giriniz.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Category? category = _categoryService.GetCategoryById(categoryId);

    if (category == null)
    {
        ConsoleRestaurantDesign.WriteError(
            "Silinecek kategori bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  ---------------------------------------------");
    Console.WriteLine("              SÝLÝNECEK KATEGORÝ");
    Console.WriteLine("  ---------------------------------------------");
    Console.ResetColor();

    Console.WriteLine($"  Kategori ID  : {category.CategoryId}");
    Console.WriteLine($"  Kategori Adý : {category.CategoryName}");

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Red;
    Console.Write("  Bu kategoriyi silmek istediðinize emin misiniz? (E/H): ");
    Console.ResetColor();

    string? confirmation = Console.ReadLine();

    if (!string.Equals(confirmation?.Trim(), "E",
        StringComparison.OrdinalIgnoreCase))
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Kategori silme iþlemi iptal edildi.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    try
    {
        _categoryService.DeleteCategory(categoryId);
    }
    catch (Microsoft.EntityFrameworkCore.DbUpdateException)
    {
        ConsoleRestaurantDesign.WriteError(
            "Kategori silinemedi. Ýliþkili ürün kayýtlarý bulunuyor olabilir.");
    }

    ConsoleRestaurantDesign.WaitForKey();
}

public void UpdateCategory()
{
    ConsoleRestaurantDesign.WriteHeader("Kategori Güncelleme");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Güncellenecek Kategori ID: ");
    Console.ResetColor();

    string? input = Console.ReadLine();

    if (!int.TryParse(input, out int categoryId) || categoryId <= 0)
    {
        ConsoleRestaurantDesign.WriteError(
            "Geçerli bir kategori ID deðeri giriniz.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Category? category = _categoryService.GetCategoryById(categoryId);

    if (category == null)
    {
        ConsoleRestaurantDesign.WriteError(
            "Güncellenecek kategori bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  ---------------------------------------------");
    Console.WriteLine("             MEVCUT KATEGORÝ BÝLGÝLERÝ");
    Console.WriteLine("  ---------------------------------------------");
    Console.ResetColor();

    Console.WriteLine($"  Kategori ID  : {category.CategoryId}");
    Console.WriteLine($"  Kategori Adý : {category.CategoryName}");

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Yeni Kategori Adý: ");
    Console.ResetColor();

    string? newCategoryName = Console.ReadLine();

    if (string.IsNullOrWhiteSpace(newCategoryName))
    {
        ConsoleRestaurantDesign.WriteError(
            "Yeni kategori adý boþ býrakýlamaz.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    string oldCategoryName = category.CategoryName;

    category.CategoryName = newCategoryName.Trim();

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- GÜNCELLEME ÖZETÝ --------");
    Console.ResetColor();

    Console.WriteLine($"  Eski Kategori Adý : {oldCategoryName}");
    Console.WriteLine($"  Yeni Kategori Adý : {category.CategoryName}");

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Deðiþiklikleri kaydetmek istiyor musunuz? (E/H): ");
    Console.ResetColor();

    string? confirmation = Console.ReadLine();

    if (!string.Equals(
        confirmation?.Trim(),
        "E",
        StringComparison.OrdinalIgnoreCase))
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Kategori güncelleme iþlemi iptal edildi.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    _categoryService.UpdateCategory(category);

    ConsoleRestaurantDesign.WaitForKey();
}
