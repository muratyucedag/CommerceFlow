using CommerceFlow.Business.Abstract;
using CommerceFlow.ConsoleUI.Helpers;

namespace CommerceFlow.ConsoleUI.Screens
{
    public class ReportScreens
    {
        private readonly ICategoryService _categoryService;
        private readonly ICustomerService _customerService;
        private readonly IProductService _productService;
        private readonly IOrderService _orderService;

        public ReportScreens(
            ICategoryService categoryService,
            ICustomerService customerService,
            IProductService productService,
            IOrderService orderService)
        {
            _categoryService = categoryService;
            _customerService = customerService;
            _productService = productService;
            _orderService = orderService;
        }

        public void ShowGeneralStatistics()
        {
            ConsoleRestaurantDesign.WriteHeader("Genel Ýstatistikler");

            var categories = _categoryService.GetAllCategories();
            var customers = _customerService.GetAllCustomers();
            var products = _productService.GetAllProducts();
            var orders = _orderService.GetAllOrders();

            int totalCategories = categories.Count;
            int totalCustomers = customers.Count;
            int totalProducts = products.Count;
            int totalOrders = orders.Count;

            int totalStock = products.Sum(x => x.ProductStock);

            int lowStockProducts =
                products.Count(x => x.ProductStock < 10);

            int outOfStockProducts =
                products.Count(x => x.ProductStock == 0);

            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine("  -------- GENEL VERÝLER --------");
            Console.ResetColor();

            Console.WriteLine($"  Toplam Kategori       : {totalCategories}");
            Console.WriteLine($"  Toplam Müþteri        : {totalCustomers}");
            Console.WriteLine($"  Toplam Ürün           : {totalProducts}");
            Console.WriteLine($"  Toplam Sipariþ        : {totalOrders}");

            Console.WriteLine();

            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine("  -------- STOK VERÝLERÝ --------");
            Console.ResetColor();

            Console.WriteLine($"  Toplam Stok Adedi     : {totalStock}");
            Console.WriteLine($"  Düþük Stoklu Ürün     : {lowStockProducts}");
            Console.WriteLine($"  Stoðu Tükenen Ürün   : {outOfStockProducts}");

            Console.WriteLine();

            ConsoleRestaurantDesign.WriteSuccess(
                "Genel istatistikler baþarýyla getirildi.");

            ConsoleRestaurantDesign.WaitForKey();
        }

        public void ShowPlaceholder(string title)
        {
            ConsoleRestaurantDesign.WriteHeader(title);

            ConsoleRestaurantDesign.WriteInfo(
                "Bu rapor daha sonra geliþtirilecek.");

            ConsoleRestaurantDesign.WaitForKey();
        }
    }
}

using CommerceFlow.ConsoleUI.Helpers;
using CommerceFlow.ConsoleUI.Screens;

namespace CommerceFlow.ConsoleUI.Menus
{
    public class ReportMenu
    {
        private readonly ReportScreens _reportScreens;

        public ReportMenu(ReportScreens reportScreens)
        {
            _reportScreens = reportScreens;
        }

        public void Show()
        {
            bool isRunning = true;

            while (isRunning)
            {
                ConsoleRestaurantDesign.WriteHeader(
                    "Raporlar ve Ýstatistikler");

                ConsoleRestaurantDesign.WriteMenuItem(
                    1, "Genel Ýstatistikler");

                ConsoleRestaurantDesign.WriteMenuItem(
                    2, "Sipariþ Durumu Raporu");

                ConsoleRestaurantDesign.WriteMenuItem(
                    3, "Satýþ ve Ciro Raporu");

                ConsoleRestaurantDesign.WriteMenuItem(
                    4, "Stok Durumu Raporu");

                ConsoleRestaurantDesign.WriteMenuItem(
                    5, "Kategori Bazlý Ürün Raporu");

                ConsoleRestaurantDesign.WriteMenuItem(
                    0, "Ana Menüye Dön");

                Console.WriteLine();

                Console.ForegroundColor = ConsoleColor.Cyan;
                Console.Write("  Seçiminiz: ");
                Console.ResetColor();

                string? choice = Console.ReadLine();

                switch (choice)
                {
                    case "1":
                        _reportScreens.ShowGeneralStatistics();
                        break;

                    case "2":
                        _reportScreens.ShowPlaceholder(
                            "Sipariþ Durumu Raporu");
                        break;

                    case "3":
                        _reportScreens.ShowPlaceholder(
                            "Satýþ ve Ciro Raporu");
                        break;

                    case "4":
                        _reportScreens.ShowPlaceholder(
                            "Stok Durumu Raporu");
                        break;

                    case "5":
                        _reportScreens.ShowPlaceholder(
                            "Kategori Bazlý Ürün Raporu");
                        break;

                    case "0":
                        isRunning = false;
                        break;

                    default:
                        ConsoleRestaurantDesign.WriteError(
                            "Geçersiz seçim yaptýnýz.");

                        ConsoleRestaurantDesign.WaitForKey();
                        break;
                }
            }
        }
    }
}

public void ShowOrderStatusReport()
{
    ConsoleRestaurantDesign.WriteHeader("Sipariþ Durumu Raporu");

    var orders = _orderService.GetAllOrders();

    if (orders.Count == 0)
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Raporlanacak sipariþ bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    int totalOrders = orders.Count;

    int preparingOrders = orders.Count(
        x => x.OrderStatus == "Hazýrlanýyor");

    int shippingOrders = orders.Count(
        x => x.OrderStatus == "Kargoda");

    int deliveredOrders = orders.Count(
        x => x.OrderStatus == "Teslim Edildi");

    int cancelledOrders = orders.Count(
        x => x.OrderStatus == "Ýptal Edildi");

    double preparingRate = (double)preparingOrders / totalOrders * 100;
    double shippingRate = (double)shippingOrders / totalOrders * 100;
    double deliveredRate = (double)deliveredOrders / totalOrders * 100;
    double cancelledRate = (double)cancelledOrders / totalOrders * 100;

    Console.ForegroundColor = ConsoleColor.Yellow;

    Console.WriteLine(
        $"  {"SÝPARÝÞ DURUMU",-22} {"ADET",-12} {"ORAN",-12}");

    Console.WriteLine(new string('-', 50));

    Console.ResetColor();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine(
        $"  {"Hazýrlanýyor",-22} {preparingOrders,-12} %{preparingRate:N2}");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.WriteLine(
        $"  {"Kargoda",-22} {shippingOrders,-12} %{shippingRate:N2}");

    Console.ForegroundColor = ConsoleColor.Green;
    Console.WriteLine(
        $"  {"Teslim Edildi",-22} {deliveredOrders,-12} %{deliveredRate:N2}");

    Console.ForegroundColor = ConsoleColor.Red;
    Console.WriteLine(
        $"  {"Ýptal Edildi",-22} {cancelledOrders,-12} %{cancelledRate:N2}");

    Console.ResetColor();

    Console.WriteLine(new string('-', 50));
    Console.WriteLine();

    ConsoleRestaurantDesign.WriteSuccess(
        $"Toplam Sipariþ Sayýsý: {totalOrders}");

    ConsoleRestaurantDesign.WaitForKey();
}

public void ShowSalesReport()
{
    ConsoleRestaurantDesign.WriteHeader("Satýþ ve Ciro Raporu");

    var orders = _orderService.GetAllOrders();

    if (orders.Count == 0)
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Raporlanacak sipariþ bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    // Toplam sipariþ tutarý
    decimal totalOrderAmount = orders.Sum(
        x => x.OrderTotalPrice);

    // Teslim edilen sipariþlerin toplam tutarý
    decimal completedRevenue = orders
        .Where(x => x.OrderStatus == "Teslim Edildi")
        .Sum(x => x.OrderTotalPrice);

    // Hazýrlanýyor ve Kargoda olan sipariþlerin tutarý
    decimal pendingAmount = orders
        .Where(x => x.OrderStatus == "Hazýrlanýyor" ||
                    x.OrderStatus == "Kargoda")
        .Sum(x => x.OrderTotalPrice);

    // Ýptal edilen sipariþlerin tutarý
    decimal cancelledAmount = orders
        .Where(x => x.OrderStatus == "Ýptal Edildi")
        .Sum(x => x.OrderTotalPrice);

    // Ýptal edilmemiþ sipariþler
    var validOrders = orders
        .Where(x => x.OrderStatus != "Ýptal Edildi")
        .ToList();

    // Ortalama sipariþ tutarý
    decimal averageOrderAmount = validOrders.Count > 0
        ? validOrders.Average(x => x.OrderTotalPrice)
        : 0;

    // En yüksek tutarlý sipariþ
    decimal highestOrderAmount = orders.Max(
        x => x.OrderTotalPrice);

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- SATIÞ VE CÝRO BÝLGÝLERÝ --------");
    Console.ResetColor();

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Toplam Sipariþ Tutarý      : ");
    Console.ForegroundColor = ConsoleColor.White;
    Console.WriteLine($"{totalOrderAmount:N2} TL");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Gerçekleþen Ciro           : ");
    Console.ForegroundColor = ConsoleColor.Green;
    Console.WriteLine($"{completedRevenue:N2} TL");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Bekleyen Sipariþ Tutarý    : ");
    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine($"{pendingAmount:N2} TL");

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Ýptal Edilen Sipariþ Tutarý: ");
    Console.ForegroundColor = ConsoleColor.Red;
    Console.WriteLine($"{cancelledAmount:N2} TL");

    Console.ResetColor();
    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- SÝPARÝÞ ANALÝZÝ --------");
    Console.ResetColor();

    Console.WriteLine();

    Console.WriteLine(
        $"  Ortalama Sipariþ Tutarý    : {averageOrderAmount:N2} TL");

    Console.WriteLine(
        $"  En Yüksek Sipariþ Tutarý   : {highestOrderAmount:N2} TL");

    Console.WriteLine(
        $"  Toplam Sipariþ Sayýsý      : {orders.Count}");

    Console.WriteLine(
        $"  Geçerli Sipariþ Sayýsý     : {validOrders.Count}");

    Console.WriteLine();

    ConsoleRestaurantDesign.WriteSuccess(
        "Satýþ ve ciro raporu baþarýyla oluþturuldu.");

    ConsoleRestaurantDesign.WaitForKey();
}

public void ShowStockReport()
{
    ConsoleRestaurantDesign.WriteHeader("Stok Durumu Raporu");

    var products = _productService.GetAllProducts();

    if (products.Count == 0)
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Raporlanacak ürün bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    // Genel stok bilgileri
    int totalProducts = products.Count;

    int totalStock = products.Sum(
        x => x.ProductStock);

    decimal totalStockValue = products.Sum(
        x => x.ProductPrice * x.ProductStock);

    // Stok durumuna göre ürün sayýlarý
    int outOfStock = products.Count(
        x => x.ProductStock == 0);

    int lowStock = products.Count(
        x => x.ProductStock > 0 &&
             x.ProductStock < 10);

    int sufficientStock = products.Count(
        x => x.ProductStock >= 10);

    // En fazla ve en az stok bulunan ürün
    var highestStockProduct = products
        .OrderByDescending(x => x.ProductStock)
        .FirstOrDefault();

    var lowestStockProduct = products
        .OrderBy(x => x.ProductStock)
        .FirstOrDefault();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- GENEL STOK BÝLGÝLERÝ --------");
    Console.ResetColor();

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Toplam Ürün Çeþidi   : ");
    Console.ForegroundColor = ConsoleColor.White;
    Console.WriteLine(totalProducts);

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Toplam Stok Adedi    : ");
    Console.ForegroundColor = ConsoleColor.White;
    Console.WriteLine(totalStock);

    Console.ForegroundColor = ConsoleColor.Cyan;
    Console.Write("  Toplam Stok Deðeri   : ");
    Console.ForegroundColor = ConsoleColor.Green;
    Console.WriteLine($"{totalStockValue:N2} TL");

    Console.ResetColor();
    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- STOK DURUMU DAÐILIMI --------");
    Console.ResetColor();

    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Red;
    Console.WriteLine(
        $"  Stoðu Tükenen Ürünler : {outOfStock}");

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine(
        $"  Düþük Stoklu Ürünler  : {lowStock}");

    Console.ForegroundColor = ConsoleColor.Green;
    Console.WriteLine(
        $"  Yeterli Stoklu Ürünler: {sufficientStock}");

    Console.ResetColor();
    Console.WriteLine();

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- STOK ANALÝZÝ --------");
    Console.ResetColor();

    Console.WriteLine();

    if (highestStockProduct != null)
    {
        Console.WriteLine(
            $"  En Fazla Stok: {highestStockProduct.ProductName}");

        ConsoleRestaurantDesign.WriteInfo(
            $"  Stok Miktarý: {highestStockProduct.ProductStock}");
    }

    Console.WriteLine();

    if (lowestStockProduct != null)
    {
        Console.WriteLine(
            $"  En Az Stok: {lowestStockProduct.ProductName}");

        ConsoleRestaurantDesign.WriteInfo(
            $"  Stok Miktarý: {lowestStockProduct.ProductStock}");
    }

    Console.WriteLine();

    ConsoleRestaurantDesign.WriteSuccess(
        "Stok raporu baþarýyla oluþturuldu.");

    ConsoleRestaurantDesign.WaitForKey();
}

public void ShowCategoryProductReport()
{
    ConsoleRestaurantDesign.WriteHeader("Kategori Bazlý Ürün Raporu");

    var categories = _categoryService.GetAllCategories();
    var products = _productService.GetAllProducts();

    if (categories.Count == 0)
    {
        ConsoleRestaurantDesign.WriteInfo(
            "Raporlanacak kategori bulunamadý.");

        ConsoleRestaurantDesign.WaitForKey();
        return;
    }

    // Ürünleri kategori ID deðerlerine göre gruplandýralým.
    var productGroups = products
        .GroupBy(x => x.CategoryId)
        .ToDictionary(
            x => x.Key,
            x => new
            {
                ProductCount = x.Count(),
                TotalStock = x.Sum(p => p.ProductStock),
                StockValue = x.Sum(
                    p => p.ProductPrice * p.ProductStock)
            });

    Console.ForegroundColor = ConsoleColor.Yellow;

    Console.WriteLine(
        $"  {"KATEGORÝ ADI",-35} {"ÜRÜN",-8} {"STOK",-10} {"STOK DEÐERÝ",-20}");

    Console.WriteLine(new string('-', 80));

    Console.ResetColor();

    foreach (var category in categories)
    {
        bool hasProducts = productGroups.TryGetValue(
            category.CategoryId, out var group);

        int productCount = hasProducts ? group!.ProductCount : 0;
        int totalStock = hasProducts ? group!.TotalStock : 0;
        decimal stockValue = hasProducts ? group!.StockValue : 0;

        Console.ForegroundColor = ConsoleColor.Cyan;
        Console.Write($"  {category.CategoryName,-35}");

        Console.ForegroundColor = ConsoleColor.White;
        Console.Write($"{productCount,-8}");
        Console.Write($"{totalStock,-10}");

        Console.ForegroundColor = ConsoleColor.Green;
        Console.WriteLine($"{stockValue,-20:N2}");
    }

    Console.ResetColor();

    Console.WriteLine(new string('-', 80));
    Console.WriteLine();

    // Genel toplamlar
    int totalProducts = products.Count;

    int totalStockQuantity = products.Sum(
        x => x.ProductStock);

    decimal totalStockValue = products.Sum(
        x => x.ProductPrice * x.ProductStock);

    Console.ForegroundColor = ConsoleColor.Yellow;
    Console.WriteLine("  -------- GENEL TOPLAMLAR --------");
    Console.ResetColor();

    Console.WriteLine();

    Console.WriteLine($"  Toplam Kategori  : {categories.Count}");
    Console.WriteLine($"  Toplam Ürün      : {totalProducts}");
    Console.WriteLine($"  Toplam Stok      : {totalStockQuantity}");

    Console.ForegroundColor = ConsoleColor.Green;
    Console.WriteLine(
        $"  Toplam Stok Deðeri : {totalStockValue:N2} TL");

    Console.ResetColor();

    Console.WriteLine();

    ConsoleRestaurantDesign.WriteSuccess(
        "Kategori bazlý ürün raporu baþarýyla oluþturuldu.");

    ConsoleRestaurantDesign.WaitForKey();
}
